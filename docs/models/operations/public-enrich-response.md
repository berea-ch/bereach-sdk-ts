# PublicEnrichResponse

```typescript
import { PublicEnrichResponse } from "bereach/models/operations";

  source: "public",
  batchId: "<id>",
  total: 799113,
  visited: 434378,
  failed: 516928,
  skipped: 838660,
  status: "processing",
};
```
