# PublicFindPeopleResponse

Found and enriched people

## Example Usage

```typescript
import { PublicFindPeopleResponse } from "bereach/models/operations";

let value: PublicFindPeopleResponse = {
  source: "public",
  provider: "<value>",
  query: "<value>",
  urlsFound: 169312,
  newUrls: 528528,
  hydrated: 487109,
  contacts: [
    {
      name: "<value>",
      headline: "<value>",
      company: "Gerhold - Kautzer",
      location: null,
      profileUrl: "https://lean-tusk.biz",
      completeness: "<value>",
    },
  ],
};
```
