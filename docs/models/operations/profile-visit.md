# ProfileVisit

Limits for visiting LinkedIn profiles and company pages

## Example Usage

```typescript
import { ProfileVisit } from "bereach/models/operations";

let value: ProfileVisit = {
  uncapped: true,
  daily: {
    current: 293170,
    limit: 692882,
    remaining: 96344,
  },
  weekly: {
    current: 791982,
    limit: 483792,
    remaining: 394394,
  },
  minIntervalSeconds: 4010,
  nextResetDaily: "<value>",
  nextResetWeekly: "<value>",
};
```
