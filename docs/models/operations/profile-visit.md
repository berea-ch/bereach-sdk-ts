```typescript
import { ProfileVisit } from "bereach/models/operations";

  uncapped: true,

  },

    current: 791982,
    limit: 483792,
    remaining: 394394,
  },
  minIntervalSeconds: 4010,
  nextResetDaily: "<value>",
  nextResetWeekly: "<value>",
};
```
